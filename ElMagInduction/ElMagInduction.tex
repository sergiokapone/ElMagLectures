\begin{frame}{Закон електромагнітної індукції}{}
	\begin{block}{Закон електромагнітної індукції (Фарадей)}\justifying
		ЕРС, що виникає в замкненому контурі, пропорційна швидкості зміни магнітного потоку крізь поверхню, обмежену цим контуром:
		\begin{equation*}
			\tcbhighmath{\EMF_\text{інд} = -\frac1c \frac{d\Phi}{dt}}, \qquad \Phi = \iint\limits_S \Bfield\cdot d\vect{S}.
		\end{equation*}
	\end{block}
	\begin{block}{}\justifying
		Магнітний потік може змінюватися:
		\begin{itemize}
			\item через \alert{рух або деформацію контуру} в постійному полі (зміна площі, поворот рамки);
			\item через \alert{зміну магнітного поля} в часі, коли контур нерухомий.
		\end{itemize}
		Фізичні причини ЕРС у цих двох випадках --- різні, хоча закон записується однаково.
	\end{block}
	\begin{block}{}\justifying\small
		Знак <<$-$>> виражає \alert{правило Ленца}. Для котушки з $N$ однакових витків під знаком похідної стоїть \alert{потокозчеплення} $\Psi = N\Phi$.
	\end{block}
\end{frame}

\subsection{Явище електромагнітної індукції в рухомих провідниках}
%% --------------------------------------------------------


% ============================== Слайд ## ===================================
\begin{frame}{Виникнення ЕРС за рахунок зміни площі контура}{}\small
	\begin{columns}[T]
		\begin{column}{0.4\linewidth}\centering
			\begin{pict}
				\includegraphics[width=0.85\linewidth]{Area_change}
			\end{pict}
		\end{column}
		\begin{column}{0.6\linewidth}
			\begin{overprint}
				\onslide<1>
				\begin{block}{За законом Фарадея}\justifying
					Поле $\Bfield$ однорідне й перпендикулярне до площини контуру. Перемичка $bc$ довжиною $\ell$ рухається зі швидкістю $v$, площа
					контуру змінюється як
					\begin{equation*}
						\frac{dS}{dt} = -\ell v,
						\qquad
						\EMF_\text{інд} = -\frac{B}{c}\frac{dS}{dt} = \frac{B\ell v}{c}.
					\end{equation*}
				\end{block}
				\onslide<2>
				\begin{block}{Через силу Лоренца}\justifying
					На електрони в рухомій перемичці діє сила $\vect{F} = -\frac{e}{c}\left[\vect{v}\times\Bfield\right]$. Її робота з перенесення
					одиничного заряду вздовж перемички і є ЕРС:
					\begin{equation*}
						\EMF_\text{інд} = \frac1c\int\limits_b^c \left[\vect{v}\times\Bfield\right]\cdot d\vect{\ell} = \frac{B\ell v}{c}.
					\end{equation*}
				\end{block}
			\end{overprint}
		\end{column}
	\end{columns}
	\begin{overprint}
		\onslide<1>
		\begin{alertblock}{}\justifying
			Але ж магнітна сила роботи не виконує! Звідки береться енергія індукційного струму?
		\end{alertblock}
		\onslide<2>
		\begin{block}{Енергетичний баланс}\justifying
			Коли тече струм $I$, на перемичку діє сила Ампера $F_A = \frac1c I\ell B$, спрямована проти руху. Щоб рух був рівномірним, зовнішня сила
			має її зрівноважувати, розвиваючи потужність
			\begin{equation*}
				P = F_A v = \frac{I\ell B v}{c} = \EMF_\text{інд} I.
			\end{equation*}
			\alert{Магнітне поле лише <<перенаправляє>> роботу зовнішніх сил}, які й постачають енергію індукційного струму.
		\end{block}
	\end{overprint}
\end{frame}
% ===========================================================================



% ============================== Слайд ## ===================================
\begin{frame}{Принцип роботи генераторів змінного струму}{}
	\begin{block}{}\justifying
		Явище електромагнітної індукції використовують у генераторах електричної енергії, без яких неможливо уявити сучасну електроенергетику. У
		генераторі механічна енергія перетворюється на електричну.
	\end{block}
	\begin{columns}
		\begin{column}{0.4\linewidth}\centering
			\begin{pict}
				\includegraphics[width=0.9\linewidth]{AC-Generator}
			\end{pict}
		\end{column}
		\begin{column}{0.6\linewidth}
			\begin{block}{}\justifying
				Найпростіший генератор --- рамка площею $S$, що обертається з кутовою швидкістю $\omega$ в однорідному полі $\Bfield$. Кінці рамки
				приєднані до кілець, до яких притиснуті щітки --- ковзні контакти. Потік крізь рамку
				\begin{equation*}
					\Phi = BS\cos\omega t,
					\qquad
					\frac{d\Phi}{dt} = -\omega BS\sin\omega t.
				\end{equation*}
			\end{block}
		\end{column}
	\end{columns}
	\begin{block}{}\justifying
		У рамці виникає змінна ЕРС, а при підключенні навантаження $R$ --- змінний струм:
		\begin{equation*}
			\EMF_\text{інд} = \frac{\omega BS}{c}\sin\omega t,
			\qquad
			I = \frac{\EMF_\text{інд}}{R} = \frac{\omega BS}{cR}\sin\omega t.
		\end{equation*}
		Частота струму дорівнює частоті обертання рамки, а амплітуда ЕРС пропорційна $\omega$.
	\end{block}
\end{frame}

\begin{frame}{Закон збереження магнітного потоку}{}\small
	\begin{block}{}
		Нехай замкнений виток з опором $R$ перебуває в зовнішньому магнітному полі.
	\end{block}
	\begin{columns}
		\begin{column}{0.2\linewidth}\centering
			\begin{tikzpicture}[>=latex, scale=0.6, transform shape]
				\begin{scope}

								\draw [in=135, out=-90, <-] (0, 2) to (0.25, 1);
								\draw [in=45, out=-45]  (0.25, 1) to (0.75, -0.5);

					\draw[->, thick] (-0.7, 0) -- ++(0.5, 0);
				\end{scope}
			\end{tikzpicture}
		\end{column}
		\begin{column}{0.8\linewidth}
			\begin{block}{}\justifying
				Будь-яка зміна магнітного поля збуджує у витку ЕРС індукції, а отже, струм
				\begin{equation*}
					I = \frac{\EMF_\text{інд}}{R} = - \frac{1}{cR} \frac{d\Phi}{dt}.
				\end{equation*}
				Якщо опір контуру $R \to 0$ (надпровідник), то
				\begin{equation*}
					\frac{d\Phi}{dt} = -cRI \to 0\ \Rightarrow\ \Phi = \const,
				\end{equation*}
				інакше навіть малі зміни $\Phi$ викликали б нескінченні струми. Отже, \alert{магнітний потік крізь контур з дуже малим опором
					зберігається}. Тут $\Phi$ --- повний потік, разом із потоком поля самого індукційного струму.
			\end{block}
		\end{column}
	\end{columns}
	\begin{block}{}\justifying
		Число силових ліній, що пронизують виток, незмінне: силові лінії ніби <<вморожені>> в провідний контур і при його зміщенні рухаються разом
		із ним. Те саме відбувається в добре провідній плазмі (<<вморожене>> поле в астрофізиці).
	\end{block}
\end{frame}

\subsection{Вихрове електричне поле}
%% --------------------------------------------------------



% ============================== Слайд ## ===================================
\begin{frame}{Вихрове електричне поле}{}
	\begin{onlyenv}<1>
		\begin{block}{}\justifying
			Оскільки $\Phi = \iint\limits_S \Bfield\cdot d\vect{S}$, а ЕРС --- це циркуляція сторонньої сили на одиничний заряд, $\EMF =
				\oint\limits_L \Efield\cdot d\vect{\ell}$, то для нерухомого контуру із закону індукції випливає:
			\begin{equation*}
				\tcbhighmath{\oint\limits_L \Efield\cdot d\vect{\ell} = - \frac1{c} \iint\limits_S \frac{\partial \Bfield}{\partial t}\cdot d\vect{S}.}
			\end{equation*}
		\end{block}
		\begin{block}{Теорема про циркуляцію $\Efield$}\justifying
			Циркуляція напруженості електричного поля по довільному контуру дорівнює взятому з протилежним знаком потоку швидкості зміни
			магнітної індукції крізь довільну поверхню, обмежену цим контуром.
		\end{block}
		\begin{block}{}\justifying
			За теоремою Стокса це рівняння можна переписати в диференціальній формі:
			\begin{equation*}
				\tcbhighmath{\Rot\Efield = - \frac1c \frac{\partial\Bfield}{\partial t}.}
			\end{equation*}
		\end{block}
	\end{onlyenv}
	\begin{onlyenv}<2>
		\begin{columns}
			\begin{column}{0.4\linewidth}\centering
				\begin{pict}
					\begin{tikzpicture}[rotate=90, scale=0.9, >=latex, midarrow/.style={%
									postaction={ decorate,
											decoration={ markings, mark=at position .7 with {\arrow{latex}}}}}]
						\begin{scope}[xshift=0.3cm]
							\draw[arrowpos={0.5}{2pt}{4pt}, red] (1.1,0) [partial ellipse=360:0:0.15 and 0.5];
							\draw[arrowpos={0.5}{2pt}{4pt}, red] (1.1,0) [partial ellipse=360:0:0.3 and 1];
							\draw[arrowpos={0.5}{2pt}{4pt}, red] (1.1,0) [partial ellipse=360:0:0.6 and 2] node[pos=0.5, below] {$\Efield$};
						\end{scope}

						\foreach \y in {-3,...,3}{
								\draw[blue, midarrow] plot[domain=1:3] (\x, 0.2*\y+0.05*\y*\x^2);
							}
						\node[blue, font=\small] at (3.5, 0) {$\frac{\partial\Bfield}{\partial t}>0$};
					\end{tikzpicture}
				\end{pict}
			\end{column}
			\begin{column}{0.6\linewidth}
				\begin{block}{}\justifying
					\begin{equation*}
						\tcbhighmath{\Rot\Efield = - \frac1c \frac{\partial\Bfield}{\partial t}}
					\end{equation*}
					За Максвеллом \alert{явище електромагнітної індукції} полягає в тому, що змінне магнітне поле збуджує в просторі електричне поле;
					провідники для цього не потрібні. Індукційні струми в провідниках --- лише наслідок дії цього поля на заряди.
				\end{block}
			\end{column}
		\end{columns}
		\begin{block}{}\justifying
			В електростатиці $\Rot\Efield = 0$, а в змінному в часі магнітному полі $\Rot\Efield \neq 0$. Індуковане електричне поле
			\alert{не є потенціальним}: його силові лінії замкнені, це \alert{вихрове} поле. Робота такого поля по замкненому контуру не дорівнює
			нулю, тому для нього не можна ввести потенціал.
		\end{block}
	\end{onlyenv}
\end{frame}

			і ніде не закінчуються, являючи собою замкнені лінії, подібні до силових ліній магнітного поля. Це так зване вихрове поле.

			електронів до швидкостей, близьких до швидкості світла. На цьому принципі ґрунтується прискорювач електронів --- бетатрон.

			запитання: \alert{а чому, власне, це поле називається електричним}? Адже воно має інше походження та іншу конфігурацію, ніж статичне
			електричне поле. Відповідь проста: \alert{вихрове поле діє на заряд точно так само, як і електростатичне, а це ми вважали і вважаємо
				головною властивістю поля}.
		}\par}

\begin{frame}{Явище самоіндукції}{}
	\begin{block}{}\justifying
		Зміна струму в контурі змінює його власне магнітне поле, а отже, і потік крізь цей самий контур. У контурі виникає ЕРС індукції. Це
		явище називають \alert{самоіндукцією}.
	\end{block}
	\begin{columns}
		\begin{column}{0.35\linewidth}\centering
			\begin{pict}
				\begin{tikzpicture}[rotate=90, scale=0.8, >=latex, midarrow/.style={%
								postaction={ decorate,
										decoration={ markings, mark=at position .7 with {\arrow{latex}}}}}]
					\begin{scope}
						\draw[arrowpos={0.5}{2pt}{7pt}, red!40, ultra thick] (1.1,0) [partial ellipse=0:360:0.3 and 1] node[pos=0.5, below] {$I$};
					\end{scope}
					\foreach \y in {-3,...,3}{
							\draw[blue, midarrow] plot[domain=1:3] (\x, 0.2*\y+0.05*\y*\x^2);
						}
				\end{tikzpicture}
			\end{pict}
		\end{column}
		\begin{column}{0.64\linewidth}
			\begin{block}{}\justifying
				Якщо поблизу контуру немає феромагнетиків, поле $\Bfield$, а отже, і повний магнітний потік $\Phi$ крізь контур пропорційні силі
				струму $I$:
				\begin{equation*}
					\Phi = \frac1c L I,
				\end{equation*}
				$L$ --- \alert{індуктивність контуру}; вона залежить лише від форми, розмірів контуру та від $\mu$ середовища.
			\end{block}
		\end{column}
	\end{columns}
	\begin{block}{}\justifying
		При зміні сили струму за законом Фарадея виникає ЕРС самоіндукції:
		\begin{equation*}
			\tcbhighmath{\EMF_\text{si} = -\frac1c\frac{d\Phi}{dt} = - \frac{L}{c^2}\frac{dI}{dt}.}
		\end{equation*}
	\end{block}
	\begin{alertblock}{}\justifying\scriptsize
		Знак мінус --- правило Ленца: ЕРС самоіндукції протидіє струму, коли він зростає, і підтримує його, коли він спадає. Індуктивність --- міра
		<<інертності>> кола щодо зміни струму, подібно до маси в механіці.
	\end{alertblock}
\end{frame}

\section{Взаємна індукція}

% ============================== Слайд ## ===================================
\begin{frame}{Взаємна індукція}{}
	\begin{columns}
		\begin{column}{0.4\linewidth}\centering
			\begin{pict}\centering
				\input{tikz/12contours.tikz}
			\end{pict}
		\end{column}
		\begin{column}{0.59\linewidth}
			\begin{overprint}
				\onslide<1>
				\begin{block}{}\justifying
					Нехай два нерухомі контури $1$ і $2$ розташовані поблизу один одного. Струм $I_1$ у контурі $1$ створює крізь контур $2$ потік,
					пропорційний $I_1$:
					\begin{equation*}
						\Phi_{21} = \frac1c L_{21}I_1.
					\end{equation*}
					Аналогічно, струм $I_2$ створює крізь контур $1$ потік
					\begin{equation*}
						\Phi_{12} = \frac1c L_{12}I_2.
					\end{equation*}
				\end{block}
				\onslide<2>
				\begin{block}{}\justifying
					Закони Ома для контурів ($L_{11} \equiv L_1$, $L_{22} \equiv L_2$):
					\begin{equation*}
						\begin{cases}
							I_1R_1 = \EMF_1 - \dfrac{L_{11}}{c^2}\dfrac{dI_1}{dt} - \dfrac{L_{12}}{c^2}\dfrac{dI_2}{dt},
							\\[1.5ex]
							I_2R_2 = \EMF_2 - \dfrac{L_{21}}{c^2}\dfrac{dI_1}{dt} - \dfrac{L_{22}}{c^2}\dfrac{dI_2}{dt}.
						\end{cases}
					\end{equation*}
					Будь-яка зміна струму в одному контурі збуджує ЕРС в іншому. Це явище називають \alert{взаємною індукцією}.
				\end{block}
			\end{overprint}
		\end{column}
	\end{columns}
	\begin{overprint}
		\onslide<1>
		\begin{block}{}\justifying
			Коефіцієнти $L_{12}$ і $L_{21}$ називають \alert{коефіцієнтами взаємної індуктивності} контурів. Вони залежать від форми, розмірів,
			взаємного розташування контурів і від $\mu$ середовища.
		\end{block}
		\onslide<2>
		\begin{block}{}\justifying
			$L_{12}$ --- алгебраїчна величина: її знак залежить від вибору додатних напрямків обходу контурів, а для ортогонально розташованих
			контурів вона може дорівнювати нулю.
		\end{block}
	\end{overprint}
\end{frame}

% ============================== Слайд ## ===================================
\begin{frame}{Розрахунок коефіцієнтів взаємної індуктивності}{}\small
	\begin{exampleblock}{Приклад}\justifying
		Довгий парамагнітний циліндр довжиною $l$ і перерізом $S$ ($V = lS$) має дві обмотки, одна поверх іншої: $n_1$ і $n_2$ витків на
		одиницю довжини ($N_i = n_i l$). Знайдіть взаємну індуктивність, нехтуючи крайовими ефектами.
	\end{exampleblock}

	\begin{overprint}
		\onslide<1>
		\begin{block}{Розв'язок}\justifying
			За означенням $L_{21} = c\,\Phi_{21}/I_1$: пропускаємо струм $I_1$ в обмотці $1$ і знаходимо повний потік крізь усі витки обмотки
			$2$. Поле обмотки $1$ і потік:
			\begin{equation*}
				B_1 = \frac{4\pi}{c}\,\mu n_1 I_1,
				\qquad
				\Phi_{21} = N_2 B_1 S = \frac{4\pi}{c}\,\mu N_2 n_1 S\, I_1.
			\end{equation*}
			Отже,
			\begin{equation*}
				L_{21} = 4\pi\mu\,\frac{N_1 N_2}{l}\,S = 4\pi\mu\, n_1 n_2 V.
			\end{equation*}
			Так само, пропускаючи струм в обмотці $2$, знаходимо $L_{12} = 4\pi\mu\, n_1 n_2 V = L_{21}$.
		\end{block}
		\onslide<2>
		\begin{block}{Зауваження}\justifying
			Порівнюючи з індуктивністю соленоїда $L_i = 4\pi\mu n_i^2 V$, бачимо, що тут
			\begin{equation*}
				L_{21} = L_{12} = \sqrt{L_1L_2}.
			\end{equation*}
			Це співвідношення виконується тому, що ми знехтували розсіюванням потоку (весь потік однієї обмотки пронизує витки іншої) і що $\mu$
			осердя не залежить від поля.

			\smallskip

			Для феромагнітного осердя $\mu = \mu(H)$: при $N_1 \neq N_2$ однакові струми створюють різні поля, і вирази для $L_{21}$ та $L_{12}$
			містять різні $\mu$, тож $L_{21} \neq L_{12}$. Власне, для нелінійного середовища коефіцієнти індуктивності втрачають сенс сталих.
		\end{block}
	\end{overprint}
\end{frame}

% ============================== Слайд ## ===================================
\begin{frame}{Теорема взаємності}{}
	\begin{block}{Теорема взаємності}\justifying
		ЕРС, що наводиться в контурі $1$, коли струм у контурі $2$ змінюється з певною швидкістю, дорівнює ЕРС, яка наводилася б у контурі $2$,
		якби той самий змінний струм протікав у контурі $1$:
		\begin{equation*}
			\EMF_1 = -\frac{L_{12}}{c^2}\frac{d{\color{red}I}}{dt} \;{\color{red}=}\; \EMF_2 = -\frac{L_{21}}{c^2}\frac{d{\color{red}I}}{dt}.
		\end{equation*}
	\end{block}
	\begin{block}{Наслідок}\justifying
		Коефіцієнти взаємної індуктивності контурів однакові (у середовищі без феромагнетиків):
		\begin{equation*}
			\tcbhighmath{L_{12} = L_{21}.}
		\end{equation*}
		Тому можна просто говорити про взаємну індуктивність двох контурів. Доведення --- з формули Неймана:
		\(
		L_{12} = \mu\oint_1\oint_2 \frac{d\vect{\ell}_1\cdot d\vect{\ell}_2}{r_{12}},
		\)
		симетричної щодо $1 \leftrightarrow 2$.
	\end{block}
	\begin{block}{Практичне застосування}\justifying\small
		Потік крізь контур $1$ від струму $I$ у контурі $2$ дорівнює потоку крізь контур $2$ від такого самого струму в контурі $1$:
		$\Phi_{12} = \Phi_{21}$. Можна рахувати той потік, який рахувати простіше.
	\end{block}
\end{frame}

			\Phi = \pi R_2^2 \cdot \frac{2\pi I}{cR_1} = \frac{2\pi^2 R_2^2}{cR_1}\, I.
		\end{equation*}

	\end{block}
	\begin{block}{Задача 2}\justifying
		Кільце лежить у площині обертання диполя, на прямій, вздовж якої напрямлена нормаль $\vect{n}$. Якщо $\theta = \omega t$ --- кут між

		\begin{equation*}
			\Phi = \pi a^2\,\frac{2p_m\cos\omega t}{l^3},
			\qquad
			I = -\frac{1}{cR}\frac{d\Phi}{dt} = \frac{2\pi a^2 p_m\,\omega}{cRl^3}\,\sin\omega t.
		\end{equation*}
		Те саме дає теорема взаємності: поле кільця зі струмом на його осі на відстані $l$ дорівнює $\dfrac{2\pi a^2 I}{cl^3}$.
	\end{block}

\subsection{Принцип роботи трансформаторів}

% ============================== Слайд ## ===================================
\begin{frame}{Трансформатор}{}
	\framesubtitle<1>{Принцип роботи}
	\framesubtitle<2>{Теорія}
	\begin{onlyenv}<1>
		\begin{block}{}\justifying
			Трансформатор --- пристрій для підвищення або зниження змінної напруги. Його дія ґрунтується на явищі взаємної індукції.
		\end{block}
		\begin{columns}
			\begin{column}{0.4\linewidth}\centering
				\includegraphics[width=0.85\linewidth]{Transformator}
			\end{column}
			\begin{column}{0.6\linewidth}
				\begin{block}{}\justifying
					Первинна і вторинна обмотки з $N_1$ і $N_2$ витками намотані на замкнене залізне осердя. Змінна ЕРС $\EMF_1$ у первинній
					обмотці створює струм $I_1$ і змінний магнітний потік $\Phi$, зосереджений в осерді. Цей потік пронизує витки вторинної обмотки
					і збуджує в ній ЕРС взаємної індукції, а в первинній --- ЕРС самоіндукції.
				\end{block}
			\end{column}
		\end{columns}
	\end{onlyenv}
	\begin{onlyenv}<2>
		\begin{block}{}\justifying
			Закон Ома для первинної обмотки ($\Phi$ --- потік крізь один виток):
			\begin{equation*}
				\EMF_1 - \frac{N_1}{c}\frac{d\Phi}{dt} = I_1R_1.
			\end{equation*}
			Опір обмотки малий: $I_1R_1 \ll \EMF_1$, тому
			\(
			\EMF_1 \approx \frac{N_1}{c}\frac{d\Phi}{dt}.
			\)
			ЕРС у вторинній обмотці:
			\begin{equation*}
				\EMF_2 = -\frac{N_2}{c}\frac{d\Phi}{dt} = -\frac{N_2}{N_1}\, \EMF_1,\qquad \text{\scriptsize знак <<$-$>> --- протилежність фаз.}
			\end{equation*}
			Відношення $k = \dfrac{N_2}{N_1}$ називається \alert{коефіцієнтом трансформації}.
		\end{block}
		\begin{block}{}\justifying\small
			Без втрат потужність зберігається: $\EMF_1 I_1 \approx \EMF_2 I_2$.
			\begin{itemize}
				\item $k > 1$ --- підвищувальний трансформатор: збільшує напругу й зменшує струм;
				\item $k < 1$ --- знижувальний трансформатор: зменшує напругу й збільшує струм.
			\end{itemize}
			Саме тому лінії електропередачі працюють за високої напруги: втрати $I^2R$ у проводах зменшуються в $k^2$ разів.
		\end{block}
	\end{onlyenv}
\end{frame}

\section{Енергія магнітного поля}
%% --------------------------------------------------------



% ============================== Слайд ## ===================================
\begin{frame}{Енергія магнітного поля}{}
	\begin{onlyenv}<1>
		\begin{block}{}\justifying
			Провідник зі струмом створює магнітне поле, яке з'являється та зникає разом зі струмом. \alert{Магнітне поле є носієм енергії}, що
			дорівнює роботі джерела на створення струму. Аналогія з механікою: $L/c^2 \leftrightarrow m$, $I \leftrightarrow v$, $W \leftrightarrow mv^2/2$ --- індуктивність є «інерцією» струму.
		\end{block}
		\begin{columns}
			\begin{column}{0.3\linewidth}\centering
				\begin{circuitikz}[
						scale=0.7, transform shape]
					\draw
					(0,0) to[L, l=$L$] ++(0,2) to[battery2, l=$\EMF$] ++(2.5, 0) to[R, l=$R$] ++(0, -2)
					to[closing switch, l=$S$, mirror] ++(-2.5, 0) -- cycle
					;
				\end{circuitikz}
			\end{column}
			\begin{column}{0.7\linewidth}
				\begin{block}{}\justifying
					Робота джерела при перенесенні заряду $dq = I\,dt$ після замикання ключа:
					\begin{equation*}
						\delta A = \EMF\, dq = I^2R\,dt - \EMF_\text{si}I\,dt
						= \underbrace{I^2R\,dt}_\text{теплота} + \underbrace{\frac{L}{c^2}I\,dI}_\text{енергія поля}.
					\end{equation*}
				\end{block}
			\end{column}
		\end{columns}
		\begin{block}{}\justifying
			Енергія магнітного поля струму:
			\begin{equation*}
				\tcbhighmath{W = \frac{L}{c^2}\int\limits_0^I I\,dI = \frac{LI^2}{2c^2}} = \frac{I\Phi}{2c} = \frac{\Phi^2}{2L}.
			\end{equation*}
		\end{block}
	\end{onlyenv}
	\begin{onlyenv}<2-3>
		\begin{block}{}
			Виразимо енергію через характеристики самого поля.
		\end{block}
		\begin{columns}
			\begin{column}{0.25\linewidth}\centering
				\includegraphics[width=0.75\linewidth]{tikz/Solenoid.pdf}
			\end{column}
			\begin{column}{0.75\linewidth}
				\begin{block}{}
					Розглянемо однорідне поле всередині довгого соленоїда об'ємом $V = Sl$:
					\begin{equation*}
						L = \frac{4\pi\mu N^2 S}{l},
						\qquad
						B = \frac{4\pi\mu NI}{c l}.
					\end{equation*}
				\end{block}
			\end{column}
		\end{columns}
	\begin{overprint}
		\onslide<2>
		\begin{block}{}\justifying
			Енергія магнітного поля:
			\begin{equation*}
				W = \frac{LI^2}{2c^2} = \frac{(4\pi\mu N I)^2\, S l}{8\pi\mu\, c^2 l^2} = \frac{B^2}{8\pi\mu}\, V = \frac{BH}{8\pi}\, V.
			\end{equation*}
		\end{block}
		\onslide<3>
		\begin{block}{}\justifying
			Густина енергії магнітного поля (лінійне ізотропне середовище):
			\begin{equation*}
				\tcbhighmath{w = \frac{BH}{8\pi} = \frac{\mu H^2}{8\pi} = \frac{B^2}{8\pi\mu}.}
			\end{equation*}
			\alert{Магнітна енергія зосереджена там, де є поле}, --- в об'ємі соленоїда. Для довільного поля $W = \dfrac{1}{8\pi}\displaystyle\iiint
				\Bfield\cdot\Hfield\, dV$.
		\end{block}
	\end{overprint}
	\end{onlyenv}
\end{frame}

% ============================== Слайд ## ===================================
\begin{frame}{Енергія магнітного поля}{Зауваження і порядки величин}\small
	\begin{block}{Хто виконує роботу?}\justifying
		Магнітне поле роботи не виконує ($\vect{F}\perp\vect{v}$). При наростанні поля роботу над зарядами виконує \alert{вихрове електричне
		поле} (закон індукції), і саме ця робота запасається в полі. У загальному випадку
		$\dfrac{\partial w}{\partial t} = \dfrac{1}{4\pi}\,\Hfield\cdot\dfrac{\partial \Bfield}{\partial t}$.
	\end{block}
	\begin{block}{Магнітне й електричне поле}\justifying
		При $B = 1$ Тл ($10^4$ Гс) $w_\text{м} = \dfrac{B^2}{8\pi}\approx 4\cdot10^{6}$ ерг/см$^3$ $= 4\cdot 10^5$ Дж/м$^3$. Електричне поле на
		межі пробою повітря ($3\cdot10^6$ В/м) дає лише $w_\text{е}\approx 40$ Дж/м$^3$. \alert{У практично досяжних полях магнітна енергія
		приблизно в $10^4$ разів щільніша за електричну.} Тому в техніці енергію й силу <<роблять>> магнітним полем: двигуни, електромагніти,
		котушки-накопичувачі. Електростатичні двигуни майже не використовують.
	\end{block}
\end{frame}

% ============================== Слайд ## ===================================
\begin{frame}{Енергія магнітного поля}{Феромагнетики і гістерезис}\small
	\begin{onlyenv}<1>
		\begin{block}{Звідки $w = BH/8\pi$ і коли вона працює}\justifying
			Елементарна робота джерел на зміну індукції в одиниці об'єму: $\delta w = \dfrac{1}{4\pi}\,H\,dB$. У лінійному середовищі
			$B = \mu H$, тому
			\begin{equation*}
				w = \frac{1}{4\pi}\int_0^B H\,dB = \frac{BH}{8\pi}.
			\end{equation*}
			Це число однозначно визначається станом поля $(B, H)$.
		\end{block}
		\begin{block}{У феромагнетику це не так}\justifying
			Залежність $B(H)$ нелінійна й \alert{неоднозначна}: при тому самому $H$ індукція залежить від передісторії намагнічування
			(\alert{гістерезис}). Тому вираз $BH/8\pi$ втрачає зміст, а інтеграл $\frac{1}{4\pi}\int H\,dB$ залежить від \alert{шляху}, а не лише
			від кінцевого стану.
		\end{block}
	\end{onlyenv}
	\begin{onlyenv}<2>
		\begin{columns}[T]
			\begin{column}{0.42\linewidth}\centering
				\begin{tikzpicture}[scale=0.85]
					\draw[->] (-2.4,0) -- (2.5,0) node[below] {$H$};
					\draw[->] (0,-1.9) -- (0,1.9) node[left] {$B$};
					\fill[blue!15] plot[smooth] coordinates {(2,1.6) (1,1.4) (0,1.0) (-0.5,0.5) (-1,0) (-1.5,-1.0) (-2,-1.6)}
					-- plot[smooth] coordinates {(-2,-1.6) (-1,-1.4) (0,-1.0) (0.5,-0.5) (1,0) (1.5,1.0) (2,1.6)} -- cycle;
					\draw[thick,blue] plot[smooth] coordinates {(2,1.6) (1,1.4) (0,1.0) (-0.5,0.5) (-1,0) (-1.5,-1.0) (-2,-1.6)};
					\draw[thick,blue] plot[smooth] coordinates {(-2,-1.6) (-1,-1.4) (0,-1.0) (0.5,-0.5) (1,0) (1.5,1.0) (2,1.6)};
					\draw[dashed] (0,1.0) node[left] {$B_r$} -- (0.0,1.0);
					\node[below left] at (-1,0) {$-H_c$};
					\node at (0.15,0.35) {\scriptsize $S$};
				\end{tikzpicture}
			\end{column}
			\begin{column}{0.56\linewidth}
				\begin{block}{Цикл перемагнічування}\justifying
					За повний цикл робота джерел на одиницю об'єму
					\begin{equation*}
						A = \frac{1}{4\pi}\oint H\,dB = \frac{S}{4\pi},
					\end{equation*}
					де $S$ --- площа петлі. Для лінійного середовища $\oint H\,dB = 0$: енергія поля поверталася в коло. У феромагнетику
					$A > 0$: частину енергії \alert{не повернуто}, її поглинули необоротні процеси (зсув доменних стінок, переорієнтація
					доменів), і вона перейшла в \alert{теплоту}.
				\end{block}
			\end{column}
		\end{columns}
		\begin{block}{Наслідки}\justifying
			Втрати на гістерезис $\propto f\,S$. Тому осердя трансформаторів роблять з \alert{м'яких} матеріалів (вузька петля, мале $H_c$), а
			постійні магніти --- з \alert{твердих} (широка петля, велике $B_r$ і $H_c$).
		\end{block}
	\end{onlyenv}
\end{frame}

\begin{frame}{Енергія системи струмів}{}\small
	\framesubtitle<2>{Парадокс?}
	\framesubtitle<3>{Розв'язання парадоксу}
	\begin{onlyenv}<1>
	\begin{block}{}\justifying
		Для двох контурів робота джерел проти ЕРС індукції (самоіндукції і взаємної) при наростанні струмів від нуля дає
		\begin{equation*}
			\tcbhighmath{W = \frac{1}{c^2}\left(\frac{L_1I_1^2}{2} + L_{12}I_1I_2 + \frac{L_2I_2^2}{2}\right)}
		\end{equation*}
		У загальному випадку $N$ контурів: $W = \dfrac{1}{2c^2}\displaystyle\sum_{i,k} L_{ik}I_iI_k = \dfrac{1}{2c}\sum_i I_i\Phi_i$, де $\Phi_i$ ---
		повний потік крізь $i$-й контур.
	\end{block}
	\end{onlyenv}
	\begin{onlyenv}<1>
	\begin{columns}[T]
		\begin{column}{0.48\linewidth}
			\begin{block}{Енергія взаємодії}\justifying
				Доданок $W_{12} = \frac{1}{c^2}L_{12}I_1I_2$ може бути від'ємним. При \alert{сталих струмах} і жорстких контурах ($L_1$, $L_2$
				незмінні) від взаємного розташування залежить лише він, тому
				\begin{equation*}
					\vect{F} = \grad W_{12}.
				\end{equation*}
				Це та сама сила $F_x = +(\partial W/\partial x)_I$, яку ми виведемо в розділі про \alert{пондеромоторні сили}: контури
				зміщуються так, щоб $L_{12}$ зростало.
			\end{block}
		\end{column}
		\begin{column}{0.48\linewidth}
			\begin{block}{Коефіцієнт зв'язку}\justifying
				Енергія поля додатна за будь-яких $I_1$, $I_2$, тому квадратична форма додатно визначена:
				\begin{equation*}
					L_{12}^2 \le L_1L_2,
					\quad
					\varkappa = \frac{|L_{12}|}{\sqrt{L_1L_2}} \le 1.
				\end{equation*}
				$\varkappa$ показує, яка частка потоку одного контуру пронизує інший.
			\end{block}
		\end{column}
	\end{columns}
	\end{onlyenv}
	\begin{onlyenv}<2>
	\begin{block}{Ідеальний зв'язок ($\varkappa = 1$): курйоз}\justifying
		Спільне осердя без розсіювання: $L_i = \Lambda N_i^2$, $L_{12} = \Lambda N_1N_2$, і
		\begin{equation*}
			W = \frac{1}{2c^2}\left(\sqrt{L_1}\,I_1 + \sqrt{L_2}\,I_2\right)^2 = \frac{\Lambda}{2c^2}\,(N_1I_1 + N_2I_2)^2 .
		\end{equation*}
		Візьмемо струми з $N_1I_1 = -N_2I_2$ (так працює навантажений трансформатор). Тоді
		\begin{equation*}
			\tcbhighmath{W = 0 .}
		\end{equation*}
	\end{block}
	\begin{alertblock}{Питання до аудиторії}\justifying
		Струми течуть, у навантаження йде потужність $\EMF_1I_1 = -\EMF_2I_2 \neq 0$, а енергія магнітного поля системи \alert{дорівнює нулю}.
		Магнітного поля в осерді вже немає? Тоді як енергія передається від первинної обмотки до вторинної?
	\end{alertblock}
	\end{onlyenv}
	\begin{onlyenv}<3>
	\begin{block}{Розв'язання: нуль --- лише в межі}\justifying
		\begin{itemize}
			\item $W = 0$ означає, що \alert{сумарні ампер-витки} $N_1I_1 + N_2I_2 \to 0$. Але в трансформаторі й $\Lambda \propto \mu \to \infty$,
			тому потік $\Phi_0 = \frac{\Lambda}{c}(N_1I_1 + N_2I_2)$ має вигляд $\infty\cdot 0$ --- він \alert{скінченний і ненульовий}.
			\item Його задає напруга первинної обмотки: $\EMF_1 = \frac{N_1}{c}\,\dot\Phi_0$. Тобто поле в осерді є й \alert{змінюється}.
			\item Змінний $\Phi_0$ наводить $\EMF_2 = -\frac{N_2}{c}\,\dot\Phi_0$, яка виконує роботу над навантаженням. Енергія
			\alert{не накопичується в полі, а проходить крізь нього} (потік вектора Умова--Пойнтінга): запасена енергія намагнічування $W_\mu$
			від навантаження не залежить.
		\end{itemize}
	\end{block}
	\begin{block}{Аналогія}\justifying
		Труба з водою: витрата велика, а об'єм води в трубі сталий. При $\varkappa < 1$ частина потоку замикається повз другу котушку
		(потоки розсіювання), і $W = 0$ вже недосяжне.
	\end{block}
	\end{onlyenv}
\end{frame}

\begin{frame}{Пондеромоторні сили в магнітному полі}{}\small
	\framesubtitle<1>{Закон збереження енергії}
	\framesubtitle<2>{Випадок $I = \const$}
	\framesubtitle<3>{Випадок $\Phi = \const$}
	\begin{center}
		\begin{pict}
			\includegraphics[height=0.26\textheight]{pull_in_solenoid}
		\end{pict}
	\end{center}
	\begin{overprint}
		\onslide<1>
		\begin{block}{}\justifying
			Залізне осердя втягується в котушку. За \alert{законом збереження енергії} робота джерела струму йде на теплоту, на зміну
			магнітної енергії і на механічну роботу:
			\begin{equation*}
				\delta A_\text{дж} = \delta Q + dW + \delta A_\text{мех}.
			\end{equation*}
			З іншого боку, за законом Ома $\delta A_\text{дж} = \EMF I\,dt = \delta Q - \EMF_\text{si} I\,dt$. Звідси
			\begin{equation*}
				\tcbhighmath{- \EMF_\text{si} I\,dt = \frac{I\,d\Phi}{c} = dW + \delta A_\text{мех}.}
			\end{equation*}
		\end{block}
		\onslide<2>
		\begin{block}{}\justifying
			Нехай джерело підтримує в колі сталий струм. Тоді $d\Phi = \frac1c I\,dL$ і
			\begin{equation*}
				-\EMF_\text{si} I\,dt = \frac{I^2}{c^2}\, dL,
				\qquad
				dW = \frac{I^2}{2c^2}\, dL.
			\end{equation*}
			Половина роботи джерела йде на збільшення енергії поля, половина --- на механічну роботу: $\delta A_\text{мех} = + dW$. Сила
			втягування:
			\begin{equation*}
				\tcbhighmath{F_x = + \left( \frac{\partial W}{\partial x} \right)_{I}}
			\end{equation*}
			Зверніть увагу на знак: «потенціальна» механічна енергія при $I=\const$ дорівнює $-W$, бо джерело виконує роботу $2\,dW$, з якої $dW$
			іде в поле.
		\end{block}
		\onslide<3>
		\begin{block}{}\justifying
			Нехай незмінним залишається потік крізь котушку (наприклад, надпровідна котушка без джерела). Тоді
			\begin{equation*}
				- \EMF_\text{si} I\,dt = \frac{I\,d\Phi}{c} = 0
				\ \Rightarrow\
				\delta A_\text{мех} = - dW,
				\qquad
				\tcbhighmath{F_x = - \left( \frac{\partial W}{\partial x} \right)_{\Phi}}
			\end{equation*}
			Механічна робота виконується за рахунок енергії поля. Обидві формули дають \alert{ту саму силу}: вона визначається станом системи, а
			не тим, що відбувається з джерелом.
		\end{block}
	\end{overprint}
\end{frame}

\begin{frame}{Підйомна сила електромагніта}{}\small
	\begin{columns}
		\begin{column}{0.3\linewidth}
			\input{tikz/electromagnet.tikz}
		\end{column}
		\begin{column}{0.69\linewidth}
			\begin{block}{}\justifying
				Теорема про циркуляцію $\Hfield$ (у залізі $H = B/\mu$, у двох зазорах ширини $x$ --- $H = B$):
				\begin{equation*}
					\frac{B}{\mu}\, l + B\cdot 2x = \frac{4\pi}{c}\, NI
					\ \Rightarrow\
					B = \frac{4\pi\mu}{c}\, \frac{NI}{l + 2\mu x},
				\end{equation*}
				де $l$ --- довжина силової лінії в залізі. Повний потік крізь $N$ витків (переріз магніту $S$):
				\begin{equation*}
					\Phi = NBS = \frac{4\pi\mu}{c}\, \frac{N^2I}{l + 2\mu x}\, S.
				\end{equation*}
			\end{block}
		\end{column}
	\end{columns}
	\begin{block}{}\justifying
		При сталому струмі $dW = \frac{I\,d\Phi}{2c}$, тому
		\begin{equation*}
			F_x = \left( \frac{\partial W}{\partial x} \right)_{I} = \frac{I}{2c} \frac{\partial \Phi}{\partial x} = -\frac{4\pi\mu^2}{c^2}\frac{N^2I^2}{(l + 2\mu
				x)^2}\, S = -\frac{B^2}{8\pi}\, 2S.
		\end{equation*}
		$F_x < 0$ --- це сила притягання. Магнітний <<тиск>> $B^2/8\pi$ діє на кожен з двох полюсів. Оскільки $\mu \gg 1$, навіть дуже вузький
		зазор $2x = l/\mu$ удвічі зменшує поле, а силу --- вчетверо.
	\end{block}
\end{frame}

\begin{frame}{Електромагніт чи постійний магніт?}{}\small
	\begin{block}{}\justifying
		Сила притягання визначається \alert{лише полем у зазорі}: $F = \dfrac{B^2}{8\pi}\,S_\text{повн}$. Звідки береться $B$ --- від струму чи
		від намагніченого заліза --- для формули байдуже. Отже, великі вантажі можна підіймати й постійним магнітом.
	\end{block}
	\begin{center}
		\begin{tblr}{
				colspec = {X[l,m]X[c,m]X[c,m]},
				row{1} = {font=\bfseries},
				hline{1,2,Z} = {0.5pt},
				rowsep = 2pt
			}
			Джерело поля                        & $B$ у зазорі, Тл & Тиск $\dfrac{B^2}{8\pi}$, кгс/см$^2$ \\
			Сталевий магніт (історично)         & $\sim 0{,}3$     & $\sim 0{,}4$                          \\
			Неодимовий магніт (NdFeB)           & $\sim 1$         & $\sim 4$                              \\
			Електромагніт із залізним осердям   & $1{,}5$--$2$     & $9$--$16$                             \\
			Надпровідний електромагніт          & $\sim 10$        & $\sim 400$                            \\
		\end{tblr}
	\end{center}
	\begin{block}{}\justifying
		Поле в залізному осерді обмежене насиченням ($B_s \approx 2$ Тл): далі збільшувати його струмом майже не вдається, тож тиск $\sim 10$ кгс/см$^2$ ---
		практична межа. Приклад: два полюси по $100$ см$^2$ при $B = 1{,}5$ Тл утримують $\sim 2$ т.
	\end{block}
\end{frame}

\begin{frame}{Переваги електромагніта}{}\small
	\begin{block}{Чому ж усе-таки електромагніт?}\justifying
		\begin{itemize}
			\item \alert{Керованість.} Вантаж захоплюється й відпускається вмиканням струму; $F \propto (NI)^2$, тому силою можна плавно
			керувати (кран на металобрухті, сепаратори).
			\item \alert{Поле вище за $B_r$.} Постійний магніт обмежений залишковою індукцією $B_r$, а в електромагніті $B$ задають ампер-витки
			$NI$ і насичення осердя; надпровідна обмотка дає $10$ Тл і більше.
			\item \alert{Масштаб.} Електромагніт можна зробити дуже великим (прискорювачі, ЯМР, термоядерні установки).
		\end{itemize}
	\end{block}
	\begin{block}{Ціна цього}\justifying
		\begin{itemize}
			\item Для утримання треба постійно витрачати потужність $I^2R$ на нагрів обмотки (для надпровідника --- на охолодження), а постійний
			магніт утримує <<безкоштовно>>. Для малих утримувачів вигідніші постійні магніти.
			\item Зазор убиває силу: при $2\mu x = l$ поле вдвічі менше, а сила --- вчетверо. Тому поверхні мають бути гладкими й чистими.
		\end{itemize}
	\end{block}
	\begin{alertblock}{Історична довідка}\justifying
		У 1831 р. Джозеф Генрі показав електромагніт, що утримував $\sim 750$ фунтів ($> 300$ кг), тоді як тогочасні сталеві магніти ---
		лише кілька фунтів. Звідси й уявлення, що <<тільки електромагнітом>>.
	\end{alertblock}
\end{frame}
